def binary_search(arr, target):
    lo, hi = 0, len(arr) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1


def gcd(a, b):
    while b:
        a, b = b, a % b
    return a


print(binary_search([1, 3, 5, 7, 9, 11], 7))  # 3
print(gcd(48, 18))  # 6
